function createFloatingPetals() {
  const container = document.getElementById('daisyBg');
  const petals = ['🌼','🌸','🍃','🌿','✿'];
  for (let i = 0; i < 14; i++) {
    const el = document.createElement('div');

    el.textContent = petals[i % petals.length];
    el.style.cssText = `
      left: ${Math.random()*100}%;

      animation-duration: ${12 + Math.random()*16}s;
      animation-delay: ${-Math.random()*20}s;

}

function switchTab(tab) {
  document.querySelectorAll('.form-tab').forEach((el,i) => {
    el.classList.toggle('active', (tab==='login' && i===0) || (tab==='register' && i===1));
  });
  document.getElementById('loginForm').classList.toggle('active', tab === 'login');
  document.getElementById('registerForm').classList.toggle('active', tab === 'register');
}

function doLogin() {
  const email = document.getElementById('loginEmail').value;
  const pass = document.getElementById('loginPass').value;
  if (!email || !pass) { showToast('⚠️ Por favor completa todos los campos'); return; }
  isLoggedIn = true;
  userName = email.split('@')[0];
  closeModal('authModal');
  document.getElementById('loginBtn').textContent = '👤 ' + userName;
  showToast('🌼 ¡Bienvenido de vuelta, ' + userName + '!');
}

function doRegister() {
  const name = document.getElementById('regName').value;
  const email = document.getElementById('regEmail').value;
  const pass = document.getElementById('regPass').value;
  const terms = document.getElementById('regTerms').checked;
  if (!name || !email || !pass) { showToast('⚠️ Por favor completa todos los campos'); return; }
  if (pass.length < 8) { showToast('⚠️ La contraseña debe tener al menos 8 caracteres'); return; }
  if (!terms) { showToast('⚠️ Debes aceptar los términos y condiciones'); return; }
  isLoggedIn = true;
  userName = name.split(' ')[0];
  closeModal('authModal');
  document.getElementById('loginBtn').textContent = '👤 ' + userName;
  showToast('🌸 ¡Cuenta creada! Bienvenido a Daisy Books, ' + userName + '!');
}

function renderBooks(filter) {
  const grid = document.getElementById('booksGrid');
  if (!grid) return;
  const filtered = filter === 'Todos' ? books :
    filter === 'Gratis' ? books.filter(b => b.price === 0) :
    books.filter(b => b.category === filter);
  grid.innerHTML = filtered.map(book => bookCardHTML(book)).join('');
}

function bookCardHTML(book) {
  const priceStr = book.price === 0 ? 'Gratis 🎁' :
    useDollars ? `$${book.usd.toFixed(2)}` : `₡${book.price.toLocaleString()}`;
  const badgeHTML = book.badge ? `<span class="book-badge badge-${book.badge}">${{new:'NUEVO',sale:'OFERTA',free:'GRATIS',audio:'AUDIO'}[book.badge]}</span>` : '';
  const stars = '★'.repeat(Math.floor(book.rating)) + (book.rating % 1 ? '½' : '');
  return `
  <div class="book-card" onclick="openBook(${book.id})">
    <div class="book-cover" style="background:${coverGradient(book.id)}">
      <span style="font-size:64px">${book.emoji}</span>
      ${badgeHTML}
    </div>
    <div class="book-info">
      <div class="book-category">${book.category}</div>
      <div class="book-title">${book.title}</div>
      <div class="book-author">${book.author}</div>
      <div class="book-rating">${stars} <span>(${book.reviews.toLocaleString()})</span></div>
      <div class="book-price-row">
        <div class="book-price ${book.price===0?'free':''}">${priceStr}</div>
      </div>
      <div class="book-actions">
        ${book.price === 0
          ? `<button class="btn btn-primary btn-sm" onclick="event.stopPropagation();readBook('${book.title}')">📖 Leer</button>`
          : `<button class="btn btn-warm btn-sm" onclick="event.stopPropagation();addToCart(${JSON.stringify({emoji:book.emoji,title:book.title,author:book.author,price:book.price,usd:book.usd,type:'comprar'}).replace(/"/g,'&quot;')})">🛒 Comprar</button>
             <button class="btn btn-sky btn-sm" onclick="event.stopPropagation();lendBook('${book.title}')">📋 Prestar</button>`
        }
      </div>
    </div>
  </div>`;
}

function coverGradient(id) {
  const gradients = [
    'linear-gradient(135deg,#C8E8F5,#87CEEB)',
    'linear-gradient(135deg,#D4F5D8,#A8D8B0)',
    'linear-gradient(135deg,#F5EBC8,#E8C97A)',
    'linear-gradient(135deg,#F5C8D4,#E8A0B0)',
    'linear-gradient(135deg,#C8D4F5,#A0B0E8)',
    'linear-gradient(135deg,#F5D4C8,#E8B090)',
    'linear-gradient(135deg,#D4F5F0,#A0E8D8)',
    'linear-gradient(135deg,#F0F5C8,#D4E87A)',
    'linear-gradient(135deg,#E8C8F5,#C890E8)',
    'linear-gradient(135deg,#C8F5E8,#90E8C0)',
  ];
  return gradients[id % gradients.length];
}

function playAudio(idx) {
  const btn = document.getElementById('playBtn' + idx);
  if (playingAudio === idx) {
    btn.textContent = '▶';
    playingAudio = null;
    showToast('⏸ Reproducción pausada');
  } else {
    if (playingAudio !== null) {
      const prevBtn = document.getElementById('playBtn' + playingAudio);
      if (prevBtn) prevBtn.textContent = '▶';
    }
    btn.textContent = '⏸';
    playingAudio = idx;
    showToast('🎧 Reproduciendo: ' + audiobooks[idx].title);
  }
}

function scrubAudio(idx, e, el) {
  const rect = el.getBoundingClientRect();
  const pct = Math.max(0, Math.min(100, ((e.clientX - rect.left) / rect.width) * 100));
  document.getElementById('audioBar' + idx).style.width = pct + '%';
}

function toggleCart() {
  document.getElementById('cartSidebar').classList.toggle('open');
}

function addToCart(item) {
  if (cart.find(c => c.title === item.title)) {
    showToast('ℹ️ "' + item.title + '" ya está en tu carrito');
    return;
  }
  cart.push(item);
  updateCart();
  showToast('✅ "' + item.title + '" agregado al carrito 🛒');
  if (!document.getElementById('cartSidebar').classList.contains('open')) {
    document.getElementById('cartSidebar').classList.add('open');
    setTimeout(() => {}, 2000);
  }
}

function removeFromCart(idx) {
  const title = cart[idx].title;
  cart.splice(idx, 1);
  updateCart();
  showToast('🗑️ "' + title + '" eliminado del carrito');
}

function updateCart() {
  const badge = document.getElementById('cartBadge');
  const items = document.getElementById('cartItems');
  const foot = document.getElementById('cartFoot');
  badge.textContent = cart.length;
  if (cart.length === 0) {
    items.innerHTML = '<div class="cart-empty"><div class="cart-empty-icon">🛒</div><p>Tu carrito está vacío.<br>¡Agrega tus libros favoritos!</p></div>';
    foot.style.display = 'none';
  } else {
    items.innerHTML = cart.map((item, i) => `
      <div class="cart-item">
        <div class="cart-item-emoji">${item.emoji}</div>
        <div style="flex:1">
          <div class="cart-item-name">${item.title}</div>
          <div class="cart-item-author">${item.author}</div>
          <div class="cart-item-price">₡${item.price.toLocaleString()} <span style="font-size:12px;color:var(--text-light);font-weight:400">$${item.usd.toFixed(2)}</span></div>
          <div style="font-size:11px;color:var(--text-light);margin-top:2px">${item.type === 'audio' ? '🎧 Audiolibro' : item.type === 'prestar' ? '📋 Préstamo' : '📖 Compra'}</div>
        </div>
        <button class="remove-item" onclick="removeFromCart(${i})">✕</button>
      </div>
    `).join('');
    const total = cart.reduce((s, i) => s + i.price, 0);
    const totalUsd = cart.reduce((s, i) => s + i.usd, 0);
    document.getElementById('cartTotalText').textContent = `₡${total.toLocaleString()} ($${totalUsd.toFixed(2)})`;
    foot.style.display = 'block';
  }
}

function checkout() {
  showToast('💳 Procesando pago de ₡' + cart.reduce((s,i)=>s+i.price,0).toLocaleString() + '...');
  setTimeout(() => {
    cart = [];
    updateCart();
    document.getElementById('cartSidebar').classList.remove('open');
    showToast('✅ ¡Compra exitosa! Disfruta tus libros 🌼');
  }, 2000);
}

// SEARCH

function handleSearch(val) {
  if (val.trim().length < 2) return;
  const q = val.toLowerCase();
  const results = books.filter(b => b.title.toLowerCase().includes(q) || b.author.toLowerCase().includes(q) || b.category.toLowerCase().includes(q));
  if (results.length > 0) {
    showSection('catalog');
    const grid = document.getElementById('booksGrid');
    if (grid) grid.innerHTML = results.map(bookCardHTML).join('');
  }
}

}
